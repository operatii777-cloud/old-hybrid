## PASUL 3 — src/shared/helpers.ts (funcții reutilizabile)

```typescript
// packages/ai/src/db/helpers.ts
import { PrismaClient } from '@prisma/client';

export async function findOrCreateCategory(
  tx: Omit<PrismaClient, '$connect' | '$disconnect' | '$on' | '$transaction' | '$use' | '$extends'>,
  tenantId: string,
  name: string
): Promise<string> {
  const existing = await (tx as any).category.findFirst({
    where: { tenantId, name, deletedAt: null },
    select: { id: true }
  });
  if (existing) return existing.id;

  const created = await (tx as any).category.create({
    data: { tenantId, name, sortOrder: 0 },
    select: { id: true }
  });
  return created.id;
}

export async function captureSnapshot(
  model: string,
  where: Record<string, any>,
  tenantId: string
): Promise<any[]> {
  const { prisma } = await import('./prismaClient');
  const records = await (prisma as any)[model.toLowerCase()].findMany({
    where: { ...where, tenantId }
  });
  return records.map((r: any) => ({ model, id: r.id, prev: r }));
}

export async function previewDbOp(
  op: any,
  tenantId: string
): Promise<any[]> {
  const { prisma } = await import('./prismaClient');
  try {
    return await (prisma as any)[op.model.toLowerCase()].findMany({
      where: { ...(op.where ?? {}), tenantId },
      take: 20,
    });
  } catch {
    return [];
  }
}

export async function executeDbOp(op: any, tenantId: string): Promise<void> {
  const { prisma } = await import('./prismaClient');
  const model = (prisma as any)[op.model.toLowerCase()];
  const tenantWhere = { ...op.where, tenantId };

  switch (op.type) {
    case 'CREATE':
      await model.create({ data: { ...op.data, tenantId } });
      break;
    case 'UPDATE':
      await model.updateMany({ where: tenantWhere, data: op.data });
      break;
    case 'UPSERT':
      await model.upsert({
        where: { ...op.where, tenantId },
        create: { ...op.create, tenantId },
        update: op.update,
      });
      break;
    case 'DELETE_SOFT':
      await model.updateMany({ where: tenantWhere, data: { deletedAt: new Date() } });
      break;
    case 'BULK_UPDATE':
      await model.updateMany({
        where: tenantWhere,
        data: op.data,
        take: op.maxRecords ?? 100,
      });
      break;
  }
}

export async function getCorrectVatForProduct(productId: string): Promise<number> {
  const { prisma } = await import('./prismaClient');
  const product = await prisma.product.findUnique({
    where: { id: productId },
    select: { type: true, category: { select: { name: true } } }
  });
  if (!product) return 9;
  const name = product.category?.name?.toLowerCase() ?? '';
  if (name.includes('alcool') || name.includes('vin') || name.includes('bere')) return 19;
  if (product.type === 'SERVICE') return 19;
  return 9; // alimente = 9% în România
}

export function generateNextIngredientCode(lastCode: string | undefined): string {
  if (!lastCode) return 'ING-001';
  const num = parseInt(lastCode.replace('ING-', '')) + 1;
  return `ING-${String(num).padStart(3, '0')}`;
}

export function normalizeRo(s: string): string {
  return s.toLowerCase()
    .replace(/[ăâÂ]/g, 'a')
    .replace(/[îÎ]/g, 'i')
    .replace(/[șşŞȘ]/g, 's')
    .replace(/[țţŢȚ]/g, 't')
    .trim();
}

export function psychoRound(cents: number): number {
  const tiers = [0, 50, 99];
  const base = Math.floor(cents / 100);
  const dec  = cents % 100;
  const near = tiers.reduce((p, c) => Math.abs(c - dec) < Math.abs(p - dec) ? c : p);
  return base * 100 + near;
}

export function normalizeQty(qty: number, from: string, to: string): number {
  const map: Record<string, number> = {
    g: 1, kg: 1000, mg: 0.001,
    ml: 1, l: 1000,
    buc: 1, bucata: 1, bucăți: 1,
    lingura: 15, linguri: 15, linguriță: 5, lingurițe: 5,
    cana: 240, căni: 240,
    plic: 1, pachete: 1,
  };
  const fromN = from.toLowerCase().trim();
  const toN   = to.toLowerCase().trim();
  if (fromN === toN) return qty;
  return qty * (map[fromN] ?? 1) / (map[toN] ?? 1);
}

export function cosineSim(a: number[], b: number[]): number {
  const dot  = a.reduce((s, v, i) => s + v * b[i], 0);
  const magA = Math.sqrt(a.reduce((s, v) => s + v * v, 0));
  const magB = Math.sqrt(b.reduce((s, v) => s + v * v, 0));
  return magA && magB ? dot / (magA * magB) : 0;
}
```

## PASUL 9 — API Endpoints în apps/api/

Adaugă în Fastify API (`apps/api/src/routes/`) un nou fișier `ai.routes.ts`:

```typescript
// apps/api/src/routes/ai.routes.ts
import { FastifyInstance } from 'fastify';
import {
  runFullAudit, repairIssues, executeAiDbOperation,
  rollbackDbOperation, addToSchema, optimizeMenu,
  syncProductToAllInterfaces
} from '@repo/ai';
import { requireAuth, requireRole } from '../middleware/auth';

export async function aiRoutes(app: FastifyInstance) {

  // ── AUDIT ──────────────────────────────────────────────────────────────────
  // GET /api/ai/audit?tenantId=xxx
  // Rulează auditul complet și returnează raportul
  app.get('/ai/audit', { preHandler: [requireAuth, requireRole('MANAGER')] },
    async (req, reply) => {
      const { tenantId } = req.query as { tenantId: string };
      const report = await runFullAudit(tenantId);
      return reply.send(report);
    }
  );

  // POST /api/ai/audit/repair
  // Body: { tenantId, issueIds?: string[], dryRun?: boolean }
  // Repară toate problemele auto-fixable (sau doar cele din issueIds)
  app.post('/ai/audit/repair', { preHandler: [requireAuth, requireRole('MANAGER')] },
    async (req, reply) => {
      const { tenantId, issueIds, dryRun } = req.body as {
        tenantId: string; issueIds?: string[]; dryRun?: boolean;
      };
      const report  = await runFullAudit(tenantId);
      const targets = issueIds
        ? report.issues.filter(i => issueIds.includes(i.id))
        : report.issues;
      const results = await repairIssues(tenantId, req.user.id, targets, { dryRun });
      return reply.send({ results, summary: {
        fixed:       results.filter(r => r.status === 'FIXED').length,
        failed:      results.filter(r => r.status === 'FAILED').length,
        needsReview: results.filter(r => r.status === 'NEEDS_REVIEW').length,
      }});
    }
  );

  // ── MENU OPTIMIZE ──────────────────────────────────────────────────────────
  // POST /api/ai/menu/optimize
  // Body: { tenantId, locationId? }
  app.post('/ai/menu/optimize', { preHandler: [requireAuth, requireRole('MANAGER')] },
    async (req, reply) => {
      const { tenantId, locationId } = req.body as { tenantId: string; locationId?: string };
      const result = await optimizeMenu(tenantId, locationId);
      return reply.send(result);
    }
  );

  // ── DB WRITE ───────────────────────────────────────────────────────────────
  // POST /api/ai/db/write
  // Body: { tenantId, instruction: string, dryRun?: boolean }
  app.post('/api/ai/db/write', { preHandler: [requireAuth, requireRole('ADMIN')] },
    async (req, reply) => {
      const { tenantId, instruction, dryRun } = req.body as {
        tenantId: string; instruction: string; dryRun?: boolean;
      };
      const result = await executeAiDbOperation(tenantId, req.user.id, instruction, { dryRun });
      return reply.send(result);
    }
  );

  // POST /api/ai/db/rollback
  // Body: { tenantId, rollbackKey: string }
  app.post('/api/ai/db/rollback', { preHandler: [requireAuth, requireRole('ADMIN')] },
    async (req, reply) => {
      const { tenantId, rollbackKey } = req.body as { tenantId: string; rollbackKey: string };
      const result = await rollbackDbOperation(tenantId, req.user.id, rollbackKey);
      return reply.send(result);
    }
  );

  // ── SCHEMA ─────────────────────────────────────────────────────────────────
  // POST /api/ai/schema/add
  // Body: { description: string, targetModel?: string, dryRun?: boolean, autoMigrate?: boolean }
  // DOAR pentru SUPER_ADMIN — schimbă schema DB
  app.post('/api/ai/schema/add', { preHandler: [requireAuth, requireRole('SUPER_ADMIN')] },
    async (req, reply) => {
      const { description, targetModel, dryRun, autoMigrate } = req.body as {
        description: string; targetModel?: string; dryRun?: boolean; autoMigrate?: boolean;
      };
      const result = await addToSchema(description, { targetModel, dryRun, autoMigrate });
      return reply.send(result);
    }
  );

  // ── SYNC ───────────────────────────────────────────────────────────────────
  // POST /api/ai/sync/product
  // Body: { tenantId, productId, action: 'ACTIVATE'|'UPDATE'|'DEACTIVATE' }
  app.post('/api/ai/sync/product', { preHandler: [requireAuth, requireRole('MANAGER')] },
    async (req, reply) => {
      const { tenantId, productId, action } = req.body as {
        tenantId: string; productId: string; action: 'ACTIVATE'|'UPDATE'|'DEACTIVATE';
      };
      await syncProductToAllInterfaces(tenantId, productId, action);
      return reply.send({ synced: true, productId, action });
    }
  );

  // ── IMPORT (upload fișier + procesare) ────────────────────────────────────
  // POST /api/ai/import
  // multipart/form-data: file + tenantId + foodCost? + dryRun?
  app.post('/api/ai/import', { preHandler: [requireAuth, requireRole('MANAGER')] },
    async (req, reply) => {
      const data     = await req.file();
      const tenantId = req.body.tenantId as string;
      const dryRun   = req.body.dryRun === 'true';
      const foodCost = parseInt(req.body.foodCost ?? '30');

      if (!data) return reply.status(400).send({ error: 'Niciun fișier trimis' });

      // Salvează temporar fișierul
      const tmpPath = `/tmp/${Date.now()}_${data.filename}`;
      const fs      = await import('fs/promises');
      await fs.writeFile(tmpPath, await data.toBuffer());

      try {
        const { parseDocument }          = await import('@repo/ai');
        const { extractRecipesFromText } = await import('@repo/ai');
        const { matchIngredients }       = await import('@repo/ai');
        const { suggestProductPrice }    = await import('@repo/ai');
        const { generateProductPhoto }   = await import('@repo/ai');
        const { createProductFromRecipe }= await import('@repo/ai');

        const doc       = await parseDocument(tmpPath);
        const extracted = await extractRecipesFromText(doc.rawText, doc.sourceType);
        const results   = [];

        for (const recipe of extracted.recipes) {
          const matches = await matchIngredients(tenantId, recipe.ingredients);
          const pricing = await suggestProductPrice(
            tenantId,
            matches.filter(m => m.matchedIngredient).map(m => ({
              ingredientId: m.matchedIngredient!.id, quantity: 0, unit: 'g'
            })),
            recipe.servings, foodCost
          );
          const photo = await generateProductPhoto(
            recipe.productName, recipe.description ?? '', recipe.category, tenantId
          ).catch(() => ({ url: '', thumbnailUrl: '' }));

          if (!dryRun) {
            const r = await createProductFromRecipe(
              tenantId, recipe, matches, pricing, photo.url, req.user.id
            );
            results.push({ ...r, productName: recipe.productName });
          } else {
            results.push({ dryRun: true, productName: recipe.productName, pricing, matchSummary: {
              total: matches.length,
              new:   matches.filter(m => m.status === 'NEW').length,
              matched: matches.filter(m => m.status !== 'NEW').length,
            }});
          }
        }

        return reply.send({
          imported: results.length,
          dryRun,
          results,
          pendingApprovalUrl: `/admin/products/pending?tenantId=${tenantId}`,
        });
      } finally {
        await import('fs/promises').then(fs => fs.unlink(tmpPath).catch(() => {}));
      }
    }
  );
}
```

## REGULI IMPORTANTE PENTRU COPILOT

1. **TypeScript strict** — zero `any` neexplicitat, zero `as unknown as`
2. **Toate funcțiile async** returnează Promise-uri tipate explicit
3. **Niciodată hard delete** — întotdeauna `deletedAt: new Date()`
4. **tenantId** injectat în ORICE query Prisma — niciodată cross-tenant
5. **Try/catch** pe toate apelurile externe (OpenAI, Cloudflare, agregatori)
6. **AuditLog** creat pentru orice scriere în DB
7. **Dry-run mode** disponibil pentru toate operațiile de scriere
8. **Valori monetare** = INTEGER în cenți (nu float RON)
9. **Rollback** disponibil pentru operații DB Writer (snapshot în Redis, TTL 1h)
10. **Rate limiting** pe foto generator (Redis lock, TTL 24h per produs)
